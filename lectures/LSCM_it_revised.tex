\begin{frame}
  \frametitle{What is ERP?}

  \only<1 |handout:1>{
  \begin{itemize}
    \item \textbf{Stand for:} \alert{Enterprise} Resources Planning
    \item \textbf{Information System:} IT/IS for transaction in company
    \item \textbf{Integration:} collect, analyze, \& report all \alert{transaction}
    \item \textbf{Important:} backbone of business
    \item \textbf{Industry:} complex, retail, education
    \item \textbf{Standard Function:} order mgt, sale, demand/material planning, HR,\alert{finance}
    \item \textbf{Advance Function:} inventory control, report, document control, rule-based
    \item \textbf{Ex. Vendor:} SAP, ORACLE, Infor/Syteline, MS Dyn/ Navision, Sage, Epicor
  \end{itemize}
  }

   \begin{center}
    \only<2 |handout:0>{

  \note{
    \noindent\textbf{Purpose:} intro to ERP\par
    \noindent\textbf{Leading Question:} What can go wrong when different functions maintain separate item records?\par
    \noindent\textbf{Teaching cue:} Trace the inventory and accounting effect of one inconsistent master record.\par
    \begin{itemize}
      \item \textbf{feature:} flow of transaction (center around Finance/Tax)
      \item \textbf{Key function:} information system for \alert{integration} and  \alert{back office} operations
      \item \textbf{Key rationales:} standard business practice, strategic competitive, technology enabler
      \item \textbf{Example} article in PO
      \item \textbf{Bad Reputation:} Silver bullet, High Failure, High Cost, Ultimate Solution,
      \item \textbf{History:} originated in EU \& in manuf; 1978 SAP (GRE) released SAP/R2 (mainframe based computer) then 1994 released SAP/R3 (unix based)
      \item \textbf{Module:} Financial Supply Chain Management (FSCM), Controlling (CO), Logistics Execution (LE), Production Planning (PP), Plant Maintenance (PM), Project System (PS)
    \end{itemize}
  }
\end{frame}

      \item \textbf{Example} article in pallet/location
    \end{itemize}

\begin{frame}
  \frametitle{What is TMS?}

  \begin{itemize}
    \item \textbf{Stand for:} \alert{Transportation} Management System
    \item \textbf{Information System:} IT/IS for operational for transportation
    \item \textbf{Integration:} collect, tracking, activity, \& \alert{visualization}
    \item \textbf{Important:} strategic tool to 3PL/ LSP
    \item \textbf{Industry:} flexible workflow, high volume, require monitoring (CashVan)
    \item \textbf{Standard Function:} order \& workflow, freight costing, fleet resources mgt, scheduling
    \item \textbf{Advance Function:} BI, rate mgt/negotiation, route planning, navigation, access control, warnings
    \item \textbf{Ex. Vendor:} TM(SAP),  ORACLE TMS, Tranplace,  Manhattan TMS, Kuebix
  \end{itemize}


  \note{
    \noindent\textbf{Purpose:} intro to TMS\par
    \noindent\textbf{Leading Question:} Which transportation decisions belong in a TMS?\par
    \noindent\textbf{Teaching cue:} Distinguish route and carrier selection from picking and stock location.\par
    \begin{itemize}
      \item \textbf{feature:} status, location, people, operation, KPI
      \item \textbf{advance:} (egg tran), rate mgt for intramodal
      \item \textbf{Example} article in OD/leg of shipment
    \end{itemize}
  }
\end{frame}

  \frametitle{Choosing providers/ vendors}

    \begin{itemize}
      \item \textbf{Package:} limited target group, special functions in modules
      \item \textbf{Technology:} text-interface \textbf{VS} GUI, web-browser, shorter life cycle
      \item \textbf{Implementation:} adjustable parameters, experienced consultants
      \item \textbf{Customer Base:} recommendation, match with equipments or operations, 'best practice', \alert{site ref.}
      \item \textbf{Professional:} trustworthy, partnership, \alert{after-sale service}, stay in business
    \end{itemize}
